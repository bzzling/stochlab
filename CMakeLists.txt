cmake_minimum_required(VERSION 3.20)
project(stochlab VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(STOCHLAB_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(stochlab_core core/src/processes.cpp core/src/exchange.cpp core/src/api.cpp)
target_include_directories(stochlab_core PUBLIC core/include)
target_compile_options(stochlab_core PRIVATE -Wall -Wextra -Wpedantic)
if(STOCHLAB_SANITIZE AND NOT EMSCRIPTEN)
  target_compile_options(stochlab_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(stochlab_core PUBLIC -fsanitize=address,undefined)
endif()
if(EMSCRIPTEN)
  add_executable(stochlab_wasm core/src/wasm.cpp)
  target_link_libraries(stochlab_wasm PRIVATE stochlab_core)
  set_target_properties(stochlab_wasm PROPERTIES OUTPUT_NAME "stochlab" SUFFIX ".mjs")
  target_compile_options(stochlab_core PRIVATE -fexceptions)
  target_compile_options(stochlab_wasm PRIVATE -fexceptions)
  target_link_options(stochlab_wasm PRIVATE -O3 -fexceptions
    "-sDISABLE_EXCEPTION_CATCHING=0" "-sMODULARIZE=1" "-sEXPORT_ES6=1"
    "-sENVIRONMENT=web,worker,node" "-sALLOW_MEMORY_GROWTH=1"
    "-sMAXIMUM_MEMORY=536870912" "-sEXPORTED_FUNCTIONS=['_sl_request']"
    "-sEXPORTED_RUNTIME_METHODS=['ccall']")
else()
  add_executable(stochlab core/src/cli.cpp)
  target_link_libraries(stochlab PRIVATE stochlab_core)
  enable_testing()
  add_executable(stochlab_tests tests/native_main.cpp tests/process_tests.cpp tests/exchange_tests.cpp)
  target_link_libraries(stochlab_tests PRIVATE stochlab_core)
  add_test(NAME unit COMMAND stochlab_tests unit)
  add_test(NAME statistics COMMAND stochlab_tests statistics)
  add_test(NAME cli_brownian COMMAND stochlab simulate brownian --paths 1000 --seed 42)
  add_test(NAME cli_invalid COMMAND stochlab simulate brownian --paths -1)
  set_tests_properties(cli_invalid PROPERTIES WILL_FAIL TRUE)
endif()
